// Painel: ajudas que deixam o formulário mais agradável. Sem JS, tudo funciona.

// Confirmação antes de ações sem volta.
document.addEventListener('click', (e) => {
  const botao = e.target.closest('[data-confirmar]')
  if (botao && !confirm(botao.dataset.confirmar)) e.preventDefault()
})

// Copiar link.
document.addEventListener('click', async (e) => {
  const b = e.target.closest('[data-copiar]')
  if (!b) return
  try {
    await navigator.clipboard.writeText(b.dataset.copiar)
    const antes = b.innerHTML
    b.textContent = 'Copiado!'
    setTimeout(() => (b.innerHTML = antes), 1600)
  } catch {
    prompt('Copie o link:', b.dataset.copiar)
  }
})

// Tamanho máximo do arquivo à venda (o servidor confere de novo; src/lib/limites.ts).
// Avisa na hora em que a lojista escolhe, sem esperar minutos de envio.
const mbTexto = (bytes) => {
  const mb = bytes / 1048576
  return `${Number.isInteger(mb) ? mb : mb.toFixed(1).replace('.', ',')} MB`
}
const erroDeTamanho = (input, arquivo) => {
  const max = Number(input.dataset.maxBytes)
  if (!max || arquivo.size <= max) return ''
  const dica = input.dataset.podeSubir
    ? 'Comprima o PDF, divida em partes ou mude de plano para enviar arquivos maiores.'
    : 'Comprima o PDF ou divida em partes (ex.: volume 1 e volume 2).'
  return `O arquivo tem ${mbTexto(arquivo.size)} e o máximo do seu plano é ${input.dataset.maxTexto || mbTexto(max)}. ${dica}`
}
const mostrarErroArquivo = (input, msg) => {
  const caixa = input.closest('.campo')
  const p = caixa?.querySelector('[data-erro-arquivo]')
  caixa?.classList.toggle('campo--erro', !!msg)
  if (p) {
    p.textContent = msg
    p.hidden = !msg
  } else if (msg) alert(msg)
}
document.addEventListener('change', (e) => {
  const input = e.target
  if (!input.matches?.('input[type=file][data-max-bytes]')) return
  const arquivo = input.files?.[0]
  const msg = arquivo ? erroDeTamanho(input, arquivo) : ''
  mostrarErroArquivo(input, msg)
  if (msg) input.value = '' // não deixa um arquivo grande demais ir junto no formulário
})

// Arquivo do produto vai direto para o armazenamento, com progresso, antes de
// salvar o formulário. Assim um PDF de 90 MB não passa pela memória do servidor.
document.addEventListener('submit', (e) => {
  const form = e.target
  const url = form.dataset.envioDireto
  const input = form.querySelector('input[type=file][data-direto]')
  const destino = form.querySelector('input[name=arquivo_key]')
  if (!url || !input || !input.files?.length || !destino || typeof XMLHttpRequest === 'undefined') return
  e.preventDefault()
  e.stopImmediatePropagation()
  const arquivo = input.files[0]
  const grande = erroDeTamanho(input, arquivo)
  if (grande) {
    mostrarErroArquivo(input, grande)
    input.value = ''
    input.focus()
    return
  }
  const botao = e.submitter || form.querySelector('button:not([type=button])')
  const textoOriginal = botao?.textContent
  if (botao) botao.disabled = true
  const xhr = new XMLHttpRequest()
  xhr.open('PUT', url)
  xhr.setRequestHeader('Content-Type', arquivo.type || 'application/octet-stream')
  xhr.setRequestHeader('x-nome-arquivo', encodeURIComponent(arquivo.name))
  xhr.upload.onprogress = (ev) => {
    if (botao && ev.lengthComputable) botao.textContent = `Enviando arquivo… ${Math.round((ev.loaded / ev.total) * 100)}%`
  }
  const falhou = (msg) => {
    if (botao) {
      botao.disabled = false
      botao.textContent = textoOriginal
    }
    alert(msg || 'O envio do arquivo falhou. Confira a internet e tente de novo.')
  }
  xhr.onerror = () => falhou()
  xhr.onload = () => {
    let r = {}
    try {
      r = JSON.parse(xhr.responseText)
    } catch {}
    // 413 sem JSON = a Cloudflare cortou antes de chegar ao servidor (mais de 100 MB).
    if (xhr.status === 413 && !r.erro) r.erro = `Arquivo grande demais. O máximo do seu plano é ${input.dataset.maxTexto || '95 MB'}.`
    if (xhr.status !== 200 || !r.key) {
      if (r.erro) mostrarErroArquivo(input, r.erro)
      return falhou(r.erro)
    }
    destino.value = r.key
    input.removeAttribute('name') // o arquivo já subiu: não manda de novo
    if (botao) botao.textContent = 'Salvando…'
    form.submit()
  }
  xhr.send(arquivo)
})

// Botão de enviar trava e avisa enquanto o upload acontece.
document.addEventListener('submit', (e) => {
  const form = e.target
  const botao = e.submitter || form.querySelector('button[type=submit], button:not([type])')
  if (!botao || !form.dataset.enviando) return
  setTimeout(() => {
    botao.disabled = true
    botao.textContent = form.dataset.enviando
  }, 0)
})

// Endereço da loja: sugere a partir do nome e diz se está livre.
;(() => {
  const nome = document.querySelector('input[name=loja]')
  const slug = document.querySelector('input[name=slug]')
  const status = document.querySelector('[data-slug-status]')
  if (!slug || !status) return
  let mexeu = !!slug.value
  const limpar = (t) =>
    t.normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 40)
  let timer
  const conferir = () => {
    clearTimeout(timer)
    const v = limpar(slug.value)
    if (!v) {
      status.textContent = 'Só letras sem acento, números e hífen.'
      status.parentElement.removeAttribute('data-livre')
      return
    }
    timer = setTimeout(async () => {
      try {
        const r = await fetch(`/api/endereco?s=${encodeURIComponent(v)}`).then((r) => r.json())
        status.textContent = r.livre ? `✓ ${r.slug} está livre` : r.motivo
        status.parentElement.dataset.livre = r.livre ? 'sim' : 'nao'
      } catch {}
    }, 300)
  }
  nome?.addEventListener('input', () => {
    if (!mexeu) {
      slug.value = limpar(nome.value)
      conferir()
    }
  })
  slug.addEventListener('input', () => {
    mexeu = true
    conferir()
  })
  slug.addEventListener('blur', () => (slug.value = limpar(slug.value)))
})()

// Cores: paletas prontas, prévia ao vivo e alerta de contraste.
;(() => {
  const previa = document.querySelector('[data-previa]')
  if (!previa) return
  const marca = document.querySelector('[data-cor=marca]')
  const destaque = document.querySelector('[data-cor=destaque]')
  const aviso = previa.querySelector('[data-contraste]')
  const luz = (hex) => {
    const c = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255).map((v) => (v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4))
    return 0.2126 * c[0] + 0.7152 * c[1] + 0.0722 * c[2]
  }
  const atualizar = () => {
    previa.style.setProperty('--marca', marca.value)
    previa.style.setProperty('--destaque', destaque.value)
    const contraste = 1.05 / (luz(destaque.value) + 0.05)
    aviso.textContent = contraste < 3.5 ? 'A cor dos botões está clara demais: o texto branco fica difícil de ler no celular.' : ''
  }
  marca.addEventListener('input', atualizar)
  destaque.addEventListener('input', atualizar)
  document.querySelectorAll('[data-paleta]').forEach((b) =>
    b.addEventListener('click', () => {
      const [m, d] = b.dataset.paleta.split(',')
      marca.value = m
      destaque.value = d
      document.querySelectorAll('[data-paleta]').forEach((x) => x.classList.toggle('ativa', x === b))
      atualizar()
    }),
  )
  atualizar()
})()

// Foto grande do celular (5–10 MB) vira uma de ~300 KB antes de subir.
document.addEventListener('change', async (e) => {
  const input = e.target
  if (!input.matches('input[type=file][data-reduzir-imagem]') || !input.files?.length) return
  const max = Number(input.dataset.reduzirImagem) || 1600
  if (typeof DataTransfer === 'undefined' || typeof createImageBitmap === 'undefined') return
  const saida = new DataTransfer()
  for (const f of input.files) {
    if (!/^image\/(jpeg|png|webp)$/.test(f.type) || f.size < 400_000) {
      saida.items.add(f)
      continue
    }
    try {
      const img = await createImageBitmap(f)
      const escala = Math.min(1, max / Math.max(img.width, img.height))
      const canvas = document.createElement('canvas')
      canvas.width = Math.round(img.width * escala)
      canvas.height = Math.round(img.height * escala)
      canvas.getContext('2d').drawImage(img, 0, 0, canvas.width, canvas.height)
      const tipo = f.type === 'image/png' ? 'image/png' : 'image/jpeg'
      const blob = await new Promise((r) => canvas.toBlob(r, tipo, 0.86))
      saida.items.add(blob && blob.size < f.size ? new File([blob], f.name.replace(/\.\w+$/, tipo === 'image/png' ? '.png' : '.jpg'), { type: tipo }) : f)
    } catch {
      saida.items.add(f)
    }
  }
  input.files = saida.files
})

// Senha: botão de mostrar/esconder e aviso de Caps Lock ligado.
;(() => {
  const olho = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M2 12s3.6-7 10-7 10 7 10 7-3.6 7-10 7S2 12 2 12z"/><circle cx="12" cy="12" r="3"/></svg>'
  const fechado = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M3 3l18 18M10.6 10.6a2 2 0 0 0 2.8 2.8M9.9 5.1A10.8 10.8 0 0 1 12 5c6.4 0 10 7 10 7a17.6 17.6 0 0 1-3.2 4.2M6.6 6.6C3.9 8.4 2 12 2 12s3.6 7 10 7a10.6 10.6 0 0 0 5.4-1.5"/></svg>'
  document.querySelectorAll('input[type=password]').forEach((input) => {
    const caixa = document.createElement('div')
    caixa.className = 'campo-senha'
    input.parentNode.insertBefore(caixa, input)
    caixa.appendChild(input)
    const b = document.createElement('button')
    b.type = 'button'
    b.className = 'campo-senha__ver'
    b.setAttribute('aria-label', 'Mostrar senha')
    b.innerHTML = olho
    b.addEventListener('click', () => {
      const ver = input.type === 'password'
      input.type = ver ? 'text' : 'password'
      b.innerHTML = ver ? fechado : olho
      b.setAttribute('aria-label', ver ? 'Esconder senha' : 'Mostrar senha')
      input.focus()
    })
    caixa.appendChild(b)
    const aviso = document.createElement('p')
    aviso.className = 'caps-aviso'
    aviso.hidden = true
    aviso.textContent = 'Caps Lock está ligado'
    caixa.after(aviso)
    const caps = (e) => {
      if (e.getModifierState) aviso.hidden = !e.getModifierState('CapsLock')
    }
    input.addEventListener('keyup', caps)
    input.addEventListener('keydown', caps)
    input.addEventListener('blur', () => (aviso.hidden = true))
    // Ao enviar, a senha volta a ficar escondida (o navegador não guarda ela visível).
    input.form?.addEventListener('submit', () => (input.type = 'password'))
  })
  // E-mail: sem letra maiúscula automática nem corretor no celular.
  document.querySelectorAll('input[type=email]').forEach((i) => {
    i.setAttribute('autocapitalize', 'none')
    i.setAttribute('spellcheck', 'false')
    i.setAttribute('autocorrect', 'off')
  })
})()

// Cadastro em etapas: mostra uma de cada vez e só avança com os campos dela
// válidos. Sem JS, o formulário aparece inteiro e o servidor confere tudo.
;(() => {
  const form = document.querySelector('[data-etapas]')
  if (!form) return
  const etapas = [...form.querySelectorAll('.etapa')]
  const indicadores = [...form.querySelectorAll('[data-indicador]')]
  let atual = Number(form.dataset.etapaInicial) || 1
  form.classList.add('etapas-ligadas')
  form.noValidate = true // a validação passa a ser por etapa, aqui embaixo

  const campo = (n) => form.querySelector(`[name="${n}"]`)
  // O e-mail compara sem diferença de maiúscula; a senha, exata.
  const conferirIguais = () => {
    const e1 = campo('email'), e2 = campo('email2'), s1 = campo('senha'), s2 = campo('senha2')
    e2?.setCustomValidity(e2.value && e1.value.trim().toLowerCase() !== e2.value.trim().toLowerCase() ? 'Os dois e-mails não são iguais.' : '')
    s2?.setCustomValidity(s2.value && s1.value !== s2.value ? 'As duas senhas não são iguais.' : '')
  }
  ;['email', 'email2', 'senha', 'senha2'].forEach((n) => campo(n)?.addEventListener('input', conferirIguais))

  const mostrar = (n, focar) => {
    atual = n
    etapas.forEach((e) => (e.hidden = Number(e.dataset.etapa) !== n))
    indicadores.forEach((li) => {
      const i = Number(li.dataset.indicador)
      li.classList.toggle('atual', i === n)
      li.classList.toggle('feita', i < n)
      if (i === n) li.setAttribute('aria-current', 'step')
      else li.removeAttribute('aria-current')
    })
    if (focar) {
      const primeiro = etapas[n - 1].querySelector('.campo--erro input, input:not([type=hidden]):not([tabindex="-1"])')
      primeiro?.focus()
    }
  }

  // A etapa só vale se todos os campos dela passam na validação do navegador.
  const valida = (n) => {
    conferirIguais()
    const status = form.querySelector('[data-slug-status]')?.parentElement
    if (n === 1 && status?.dataset.livre === 'nao') {
      campo('slug').setCustomValidity(status.textContent.trim())
    } else campo('slug')?.setCustomValidity('')
    for (const i of etapas[n - 1].querySelectorAll('input')) {
      if (!i.checkValidity()) {
        i.reportValidity()
        return false
      }
    }
    return true
  }

  form.addEventListener('click', (e) => {
    if (e.target.closest('[data-avancar]') && valida(atual)) mostrar(atual + 1, true)
    if (e.target.closest('[data-voltar]')) mostrar(atual - 1, true)
  })
  // Enter num campo avança em vez de enviar o formulário pela metade.
  form.addEventListener('keydown', (e) => {
    if (e.key !== 'Enter' || e.target.tagName !== 'INPUT' || atual === etapas.length) return
    e.preventDefault()
    if (valida(atual)) mostrar(atual + 1, true)
  })
  form.addEventListener('submit', (e) => {
    for (let n = 1; n <= etapas.length; n++) {
      if (!valida(n)) {
        e.preventDefault()
        mostrar(n, false)
        valida(n)
        return
      }
    }
  })
  mostrar(atual, atual > 1)
})()
