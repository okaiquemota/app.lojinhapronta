// Apresentação: barra do topo que ganha linha ao rolar, menu do celular,
// seções que entram ao rolar, link da seção atual e o botão fixo do celular.
// Sem este arquivo a página funciona inteira (as perguntas usam <details>).
document.documentElement.classList.add('js')

;(() => {
  const navbar = document.querySelector('.navbar')
  const fixo = document.querySelector('[data-cta-fixo]')
  const capa = document.querySelector('.capa')
  const rolou = () => {
    const y = window.scrollY
    navbar?.classList.toggle('navbar--rolou', y > 8)
    if (fixo && capa) fixo.classList.toggle('cta-fixo--visivel', y > capa.offsetHeight * 0.8)
  }
  addEventListener('scroll', rolou, { passive: true })
  rolou()

  // menu do celular
  const botao = document.querySelector('.navbar__menu')
  const menu = document.getElementById('menu-celular')
  const fechar = (focar) => {
    navbar.classList.remove('navbar--aberta')
    botao.setAttribute('aria-expanded', 'false')
    botao.setAttribute('aria-label', 'Abrir menu')
    menu.hidden = true
    if (focar) botao.focus()
  }
  botao?.addEventListener('click', () => {
    if (!menu.hidden) return fechar(false)
    navbar.classList.add('navbar--aberta')
    botao.setAttribute('aria-expanded', 'true')
    botao.setAttribute('aria-label', 'Fechar menu')
    menu.hidden = false
    menu.querySelector('a')?.focus()
  })
  menu?.addEventListener('click', (e) => {
    if (e.target.closest('a')) fechar(false)
  })
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape' && menu && !menu.hidden) fechar(true)
  })

  // seções entram ao aparecer na tela
  const revelar = document.querySelectorAll('.reveal')
  if ('IntersectionObserver' in window && !matchMedia('(prefers-reduced-motion: reduce)').matches) {
    const obs = new IntersectionObserver(
      (entradas) => {
        for (const e of entradas) {
          if (!e.isIntersecting) continue
          e.target.setAttribute('data-visivel', '')
          obs.unobserve(e.target)
        }
      },
      { rootMargin: '0px 0px -6% 0px', threshold: 0.08 },
    )
    revelar.forEach((el) => obs.observe(el))
  } else {
    revelar.forEach((el) => el.setAttribute('data-visivel', ''))
  }

  // link da seção atual aceso no topo
  const links = [...document.querySelectorAll('.navbar__links a')]
  const secoes = links.map((a) => document.querySelector(a.getAttribute('href'))).filter(Boolean)
  if ('IntersectionObserver' in window) {
    const obs = new IntersectionObserver(
      (entradas) => {
        for (const e of entradas) {
          if (!e.isIntersecting) continue
          links.forEach((a) => {
            if (a.getAttribute('href') === `#${e.target.id}`) a.setAttribute('aria-current', 'true')
            else a.removeAttribute('aria-current')
          })
        }
      },
      { rootMargin: '-45% 0px -50% 0px' },
    )
    secoes.forEach((s) => obs.observe(s))
  }
})()
